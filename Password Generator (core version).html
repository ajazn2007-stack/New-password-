A sketch of the idea in Python: collect the chosen character sets, then pick characters at random.

import random, string

def generate(length, upper, lower, digits, symbols):
    pool = ""
    if upper:   pool += string.ascii_uppercase
    if lower:   pool += string.ascii_lowercase
    if digits:  pool += string.digits
    if symbols: pool += string.punctuation
    if not pool or length < 4:
        raise ValueError("Choose a length of 4+ and one type")
    return "".join(random.choice(pool)
                   for _ in range(length))
